// Process Login Petugas document.getElementById('loginForm').addEventListener('submit', async (e) => { e.preventDefault(); const btn = document.getElementById('btnLogin'); btn.innerText = "Memproses..."; btn.disabled = true; const email = document.getElementById('selectPetugas').value; const password = document.getElementById('password').value; try { const { data, error } = await supabase.auth.signInWithPassword({ email, password }); if (error) { // Menampilkan pesan error spesifik jika login gagal alert("Gagal Login: " + error.message); btn.innerText = "MASUK"; btn.disabled = false; } else { alert("Login Berhasil!"); // Pindah ke tampilan form petugas document.getElementById('loginSection').classList.add('hidden'); document.getElementById('formSection').classList.remove('hidden'); document.getElementById('userAccount').innerText = "Logged in: " + data.user.email; } } catch (err) { alert("Terjadi kesalahan sistem/koneksi: " + err.message); btn.innerText = "MASUK"; btn.disabled = false; } });
